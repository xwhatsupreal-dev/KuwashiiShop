     5	} from 'lucide-react';
     6

     8	  topupModalStep,

    10	  handleTopup,
    11	  isProcessingTopup,
    12	  angpaoCode,
    13	  setAngpaoCode,
    14	  slipFile,
    15	  setSlipFile,
    16	  topupTarget,
    17	  globalStats,

    19	  tosAccepted,
    20	  setTosAccepted,
    21	  setShowTopupTos,
    22	  
    23	}: any) => {

    37	  return (
    38	    <motion.div

    42	      className="w-full max-w-xl mx-auto py-8 flex flex-col justify-start px-4 sm:px-0"
    43	    >
    44	      <motion.div

    54	          className="absolute left-0 top-1 text-zinc-400 hover:text-white flex items-center justify-center p-2 rounded-full hover:bg-white/5 transition-colors"
